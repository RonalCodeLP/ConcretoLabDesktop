\newcommand{\portadaConcretoLab}[1]{%
	\clearpage
	\renewcommand{\thepage}{\nameportraitpage}
	\setpagemargincm{\pagemarginleft}{\firstpagemargintop}{\pagemarginright}{\pagemarginbottom}
	\fancypagestyle{portraitstyle}{%
		\fancyhf{}%
		\fancyhead[L]{%
			\COREwriteheaderitem{\nombreuniversidad}
			\COREwriteheaderitem{\nombrefacultad}
			\COREwriteheaderitem{\departamentouniversidad}
			\vspace{-0.95\baselineskip}%
		}%
		\fancyhead[R]{%
			\hspace{-0.255cm}%
			\coreinsertkeyimage{\imagendepartamentoparams}{\imagendepartamento}%
			\vspace{-0.175cm}%
		}%
	}
	\thispagestyle{portraitstyle}
	\begin{spacing}{1.025}
	~ \\
	\vfill
	\begin{center}
		\includegraphics[width=3.4cm]{marca/icono-concretolab} \\[1.1cm]
		{\fontsize{32}{38}\selectfont \bfseries \color{clazul} \titulodelinforme} \\[0.45cm]
		{\Large \color{cltexto} \temaatratar} \\[0.75cm]
		{\large \color{cltexto} #1}
	\end{center}
	\vfill
	\noindent
	\begin{minipage}{1.0\linewidth}%
		\begin{flushright}%
			\tablaintegrantes
		\end{flushright}
	\end{minipage}
	\end{spacing}
}

% Paso con su captura debajo, sin leyenda (manual directo):

\NewDocumentCommand{\pasocaptura}{m m O{0.92} m m}{%
	\par\medskip\noindent\begin{minipage}{\linewidth}
		{\bfseries #1.\enspace #2}\par\smallskip
		\centering
		\begin{tikzpicture}
			\node[anchor=south west, inner sep=0, draw=clgrisborde, line width=0.4pt] (img) at (0,0)
				{\includegraphics[width=#3\linewidth]{#4}};
			\begin{scope}[x={(img.south east)}, y={(img.north west)}]
				#5
			\end{scope}
		\end{tikzpicture}
	\end{minipage}\par\medskip}

	\node[circle, fill=clrojoborde, text=white, font=\bfseries\fontsize{6.5}{7}\selectfont, inner sep=0pt,
